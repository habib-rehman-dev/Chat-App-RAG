import { cookies } from "next/headers";
import { auth } from "@/auth";

const backendUrl = (process.env.BACKEND_URL ?? "http://localhost:8000").replace(/\/$/, "");

export async function POST(req: Request) {
  const session = await auth();
  if (!session?.user) {
    return Response.json({ detail: "Unauthorized" }, { status: 401 });
  }

  const jar = await cookies();
  const token =
    jar.get("__Secure-authjs.session-token")?.value ?? jar.get("authjs.session-token")?.value;
  if (!token) {
    return Response.json({ detail: "Missing session token" }, { status: 401 });
  }

  // Body from the frontend: { id, messages, trigger, chatId, documentIds }
  const body = await req.text();

  let upstream: Response;
  try {
    upstream = await fetch(`${backendUrl}/chat`, {
      method: "POST",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
      body,
      signal: req.signal,
    });
  } catch {
    return Response.json({ detail: "Backend is unreachable" }, { status: 502 });
  }

  // Surface upstream failures as real errors instead of an empty stream.
  if (!upstream.ok || !upstream.body) {
    const detail = (await upstream.text().catch(() => "")) || "Upstream error";
    return Response.json({ detail }, { status: upstream.status || 502 });
  }

  return new Response(upstream.body, {
    status: 200,
    headers: {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache, no-transform",
      Connection: "keep-alive",
      "x-vercel-ai-ui-message-stream": "v1",
    },
  });
}

import { cookies } from "next/headers";
import { auth } from "@/auth";

const backendUrl = (process.env.BACKEND_URL ?? "http://localhost:8000").replace(/\/$/, "");

async function getToken() {
  const jar = await cookies();
  return (
    jar.get("__Secure-authjs.session-token")?.value ??
    jar.get("authjs.session-token")?.value ??
    null
  );
}

export async function POST(req: Request) {
  const session = await auth();
  if (!session?.user) return Response.json({ detail: "Unauthorized" }, { status: 401 });

  const token = await getToken();
  if (!token) return Response.json({ detail: "Missing session token" }, { status: 401 });

  const formData = await req.formData();
  const file = formData.get("file");
  if (!(file instanceof File)) {
    return Response.json({ detail: "No file provided" }, { status: 400 });
  }

  const upstreamForm = new FormData();
  upstreamForm.append("file", file, file.name);

  let upstream: Response;
  try {
    upstream = await fetch(`${backendUrl}/api/documents/upload`, {
      method: "POST",
      headers: { Authorization: `Bearer ${token}` },
      body: upstreamForm,
      signal: req.signal,
    });
  } catch {
    return Response.json({ detail: "Backend is unreachable" }, { status: 502 });
  }

  const text = await upstream.text();
  return new Response(text, {
    status: upstream.status,
    headers: { "Content-Type": "application/json" },
  });
}

export async function GET(req: Request) {
  const session = await auth();
  if (!session?.user) return Response.json({ detail: "Unauthorized" }, { status: 401 });

  const token = await getToken();
  if (!token) return Response.json({ detail: "Missing session token" }, { status: 401 });

  let upstream: Response;
  try {
    upstream = await fetch(`${backendUrl}/api/documents`, {
      headers: { Authorization: `Bearer ${token}` },
      cache: "no-store",
      signal: req.signal,
    });
  } catch {
    return Response.json({ detail: "Backend is unreachable" }, { status: 502 });
  }

  const text = await upstream.text();
  return new Response(text, {
    status: upstream.status,
    headers: { "Content-Type": "application/json" },
  });
}

"use client";

import { useState } from "react";
import type { UIMessage } from "ai";
import { Check, Copy, FileText } from "lucide-react";
import { Button } from "@/components/ui/button";
import { cn } from "@/lib/utils";

/* ---------- minimal markdown: fenced code, `inline`, **bold** ---------- */

function Inline({ text }: { text: string }) {
  return (
    <>
      {text.split(/(`[^`\n]+`|\*\*[^*\n]+\*\*)/g).map((t, i) => {
        if (t.length > 1 && t.startsWith("`") && t.endsWith("`")) {
          return (
            <code key={i} className="rounded bg-muted px-1 py-0.5 font-mono text-[0.85em]">
              {t.slice(1, -1)}
            </code>
          );
        }
        if (t.length > 4 && t.startsWith("**") && t.endsWith("**")) {
          return <strong key={i}>{t.slice(2, -2)}</strong>;
        }
        return t;
      })}
    </>
  );
}

function RichText({ text }: { text: string }) {
  // The `$` alternative keeps a still-streaming (unclosed) fence rendering as code.
  const blocks = text.split(/(```[\s\S]*?(?:```|$))/g).filter(Boolean);
  return (
    <div className="space-y-2">
      {blocks.map((block, i) =>
        block.startsWith("```") ? (
          <pre
            key={i}
            className="overflow-x-auto rounded-lg bg-muted p-3 font-mono text-[0.85em] leading-relaxed"
          >
            <code>{block.replace(/^```[^\n]*\n?/, "").replace(/```$/, "")}</code>
          </pre>
        ) : (
          <p key={i} className="whitespace-pre-wrap wrap-break-word">
            <Inline text={block} />
          </p>
        ),
      )}
    </div>
  );
}

/* ---------- message ---------- */

interface Citation {
  key: string;
  label: string;
  href?: string;
}

function getCitations(message: UIMessage): Citation[] {
  const seen = new Set<string>();
  const out: Citation[] = [];
  for (const p of message.parts) {
    if (p.type === "source-document") {
      const label = p.filename ?? p.title;
      if (!seen.has(p.sourceId)) {
        seen.add(p.sourceId);
        out.push({ key: p.sourceId, label });
      }
    } else if (p.type === "source-url") {
      if (!seen.has(p.url)) {
        seen.add(p.url);
        out.push({ key: p.url, label: p.title ?? p.url, href: p.url });
      }
    }
  }
  return out;
}

export function ChatMessage({ message }: { message: UIMessage }) {
  const [copied, setCopied] = useState(false);
  const isUser = message.role === "user";

  const text = message.parts
    .flatMap((p) => (p.type === "text" ? [p.text] : []))
    .join("");
  const reasoning = message.parts
    .flatMap((p) => (p.type === "reasoning" ? [p.text] : []))
    .join("");
  const citations = isUser ? [] : getCitations(message);

  const copy = async () => {
    try {
      await navigator.clipboard.writeText(text);
      setCopied(true);
      setTimeout(() => setCopied(false), 1800);
    } catch {
      /* clipboard unavailable */
    }
  };

  if (!text && !reasoning && citations.length === 0) return null;

  return (
    <div className={cn("group/msg flex flex-col gap-1.5", isUser ? "items-end" : "items-start")}>
      {reasoning && (
        <details className="max-w-full text-sm text-muted-foreground">
          <summary className="cursor-pointer select-none">Thought process</summary>
          <p className="mt-1 whitespace-pre-wrap">{reasoning}</p>
        </details>
      )}

      {text && (
        <div
          className={cn(
            "max-w-full text-sm leading-relaxed",
            isUser
              ? "max-w-[85%] rounded-2xl rounded-br-md bg-primary px-3.5 py-2 text-primary-foreground"
              : "w-full",
          )}
        >
          {isUser ? (
            <p className="whitespace-pre-wrap wrap-break-word">{text}</p>
          ) : (
            <RichText text={text} />
          )}
        </div>
      )}

      {citations.length > 0 && (
        <div className="flex flex-wrap items-center gap-1.5">
          {citations.map((c, i) => {
            const chip = (
              <>
                <span className="font-medium">{i + 1}</span>
                <FileText className="size-3" />
                <span className="max-w-48 truncate">{c.label}</span>
              </>
            );
            const cls =
              "inline-flex items-center gap-1 rounded-full border bg-background px-2 py-0.5 text-xs text-muted-foreground";
            return c.href ? (
              <a key={c.key} href={c.href} target="_blank" rel="noreferrer" className={cn(cls, "hover:text-foreground")}>
                {chip}
              </a>
            ) : (
              <span key={c.key} className={cls} title={c.label}>
                {chip}
              </span>
            );
          })}
        </div>
      )}

      {!isUser && text && (
        <Button
          variant="ghost"
          size="icon-xs"
          onClick={copy}
          className="opacity-0 transition-opacity group-hover/msg:opacity-100 focus-visible:opacity-100 max-md:opacity-100"
          aria-label="Copy answer"
        >

const transport = new DefaultChatTransport({ api: "/api/chat" });

interface ChatViewProps {
  sessionId: string;
  title: string;
  initialMessages: UIMessage[];
  user: ChatUser;
  selectedDocs: DocItem[];
  onTurnFinished: (sessionId: string, messages: UIMessage[]) => void;
  onOpenSidebar: () => void;
  onRequestDelete: () => void;
}

export function ChatView({
  sessionId,
  title,
  initialMessages,
  user,
  selectedDocs,
  onTurnFinished,
  onOpenSidebar,
  onRequestDelete,
}: ChatViewProps) {
  const [input, setInput] = useState("");
  const scrollRef = useRef<HTMLDivElement>(null);
  const stickToBottom = useRef(true);
  const lastSaved = useRef(initialMessages);

  const { messages, sendMessage, status, stop, error, regenerate, setMessages } = useChat({
    id: sessionId,
    messages: initialMessages,
    transport,
  });

  const busy = status === "submitted" || status === "streaming";
  // Sent with every request so the backend can filter retrieval to these files.
  const requestBody = { session_id: sessionId, documentIds: selectedDocs.map((d) => d.id) };

  // Persist once a turn settles (not on every streamed token).
  useEffect(() => {
    if ((status === "ready" || status === "error") && messages !== lastSaved.current) {
      lastSaved.current = messages;
      onTurnFinished(sessionId, messages);
    }
  }, [status, messages, sessionId, onTurnFinished]);

  // Auto-scroll only while the reader is already near the bottom.
  useEffect(() => {
    const el = scrollRef.current;
    if (el && stickToBottom.current) el.scrollTop = el.scrollHeight;
  }, [messages, status]);

  const submit = () => {
    const text = input.trim();
    if (!text || busy) return;
    stickToBottom.current = true;
    sendMessage({ text }, { body: requestBody });
    setInput("");
  };

  return (
    <div className="flex h-full min-h-0 flex-col">
      <ChatHeader
        user={user}
        title={title}
        sourceCount={selectedDocs.length}
        hasMessages={messages.length > 0}
        onOpenSidebar={onOpenSidebar}
        onClear={() => setMessages([])}
        onDelete={onRequestDelete}
      />

      <div
        ref={scrollRef}
        onScroll={(e) => {
          const el = e.currentTarget;
          stickToBottom.current = el.scrollHeight - el.scrollTop - el.clientHeight < 80;

        className="min-h-0 flex-1 overflow-y-auto"
      >
        <div className="mx-auto flex min-h-full max-w-3xl flex-col gap-6 px-4 py-6">
          {messages.length === 0 && !error ? (
            <div className="m-auto flex max-w-sm flex-col items-center gap-2 text-center">
              <Sparkles className="size-6 text-muted-foreground" />
              <h2 className="text-base font-medium">Ask about your sources</h2>
              <p className="text-sm text-muted-foreground">
                {selectedDocs.length === 0
                  ? "Add files in the sidebar and tick the ones to include."
                  : "Answers are drawn from the files ticked in the sidebar and cite where they came from."}
              </p>
            </div>
          ) : (
            messages.map((m) => <ChatMessage key={m.id} message={m} />)
          )}

          {/* Only before the first token arrives */}
          {status === "submitted" && (
            <div className="flex items-center gap-2 text-sm text-muted-foreground">
              <Spinner /> Searching your sources…
            </div>
          )}

          {error && (
            <div className="flex items-center justify-between gap-3 rounded-lg border border-destructive/30 bg-destructive/10 p-3">
              <p className="text-sm text-destructive">{error.message || "Something went wrong."}</p>

                <RefreshCw /> Retry
              </Button>
            </div>
          )}
        </div>
      </div>

      <div className="border-t bg-background p-3">
        <form
          className="mx-auto flex max-w-3xl items-end gap-2"
          onSubmit={(e) => {
            e.preventDefault();
            submit();
          }}
        >
          <Textarea
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={(e) => {
              if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) {
                e.preventDefault();
                submit();
              }
            }}
            placeholder="Ask a question about your sources"
            rows={1}
            className="max-h-40 min-h-9 flex-1 resize-none"
            aria-label="Message"
          />

            <Button type="button" variant="destructive" size="icon-lg" onClick={() => stop()} aria-label="Stop generating">
              <Square />
            </Button>
          ) : (
            <Button type="submit" size="icon-lg" disabled={!input.trim()} aria-label="Send message">
              <ArrowUp />
            </Button>
          )}
        </form>
        <p className="mx-auto mt-1.5 max-w-3xl text-center text-xs text-muted-foreground">
          Enter to send, Shift+Enter for a new line
        </p>
      </div>
    </div>
  );
}

import { documentsApi, type RemoteDoc } from "@/features/documents/documents-api";
import { useEffect, useState } from "react";

import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from "@/components/ui/alert-dialog";
import {
  Sheet,
  SheetContent,
  SheetHeader,
  SheetTitle,
} from "@/components/ui/sheet";
import { Spinner } from "@/components/ui/spinner";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
// import { documentsApi } from "@/featuresdocuments/uploaddocuments/upload-api";
import { ChatView } from "./chat-view";
import { ChatsPanel } from "./chats-panel";
import { SourcesPanel } from "./sources-panel";
import { newId, sessionsApi } from "../sessions-api";
import { useSessions } from "../use-sessions";
import { useHydrated, useStored } from "../storage";
import type { ChatUser, DocItem, DocStatus, SessionSummary } from "../types";

const EMPTY_DOCS: DocItem[] = [];
const EMPTY_IDS: string[] = [];
const NEW_CHAT = "New chat";

type Pending = { kind: "chat" | "doc"; id: string } | null;

const errMsg = (e: unknown, fallback: string) =>
  e instanceof Error ? e.message : fallback;




export function Workspace({ user }: { user: ChatUser }) {
  // Chat sessions live on the backend. Only the file list/selection is local for now.
  const { sessions, error: sessionsError, refresh, remove } = useSessions();
  const [docs, updateDocs] = useStored<DocItem[]>(
    `rag:${user.id}:docs`,
    EMPTY_DOCS,
  );
  const [selected, updateSelected] = useStored<string[]>(
    `rag:${user.id}:selected`,
    EMPTY_IDS,
  );
  const hydrated = useHydrated();

  // First screen is an unsaved draft; the backend creates the session on its first message.
  const [boot] = useState(() => {
    const id = newId();
    return { id, initial: { [id]: [] as UIMessage[] } };
  });
  const [activeId, setActiveId] = useState(boot.id);
  // Messages each opened session starts from (fetched from the backend, or [] for a draft).
  const [initial, setInitial] = useState<Record<string, UIMessage[]>>(
    boot.initial,
  );
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [pending, setPending] = useState<Pending>(null);
  const [notice, setNotice] = useState<string | null>(null);

  const readyDocs = docs.filter((d) => d.status === "ready");
  const selectedIds = selected.filter((id) =>
    readyDocs.some((d) => d.id === id),
  );
  const selectedDocs = readyDocs.filter((d) => selectedIds.includes(d.id));

  const activeSession = sessions.find((s) => s.id === activeId);
  const isDraft = !activeSession;
  const listItems: SessionSummary[] = isDraft
    ? [{ id: activeId, title: NEW_CHAT, updatedAt: 0 }, ...sessions]
    : sessions;

  // Load history for a past session the first time it is opened.
  useEffect(() => {
    if (initial[activeId] !== undefined) return;
    let cancelled = false;
    sessionsApi
      .get(activeId)
      .then((messages) => {
        if (!cancelled) setInitial((p) => ({ ...p, [activeId]: messages }));
      })
      .catch((e) => {
        if (!cancelled)
          setNotice(`Could not open this chat: ${errMsg(e, "unknown error")}`);
      });
    return () => {
      cancelled = true;
    };
  }, [activeId, initial]);

  /* ---------- chats ---------- */

  function startFresh() {
    const id = newId();
    setInitial((p) => ({ ...p, [id]: [] }));
    setActiveId(id);
  }

  function newChat() {
    setNotice(null);
    setSidebarOpen(false);
    // Already on an empty draft: nothing to create.
    if (isDraft && (initial[activeId]?.length ?? 0) === 0) return;
    startFresh();
  }

  function openSession(id: string) {
    setNotice(null);
    setActiveId(id);
    setSidebarOpen(false);
  }

  // Called by ChatView when a turn finishes: keep the local copy fresh and pull the
  // backend's list so a new session, its title and ordering show up in the sidebar.
  function onTurnFinished(sessionId: string, messages: UIMessage[]) {
    setInitial((p) => ({ ...p, [sessionId]: messages }));
    void refresh();
  }

  /* ---------- sources ---------- */

  function setStatus(id: string, status: DocStatus, error?: string) {
    updateDocs((p) =>
      p.map((d) => (d.id === id ? { ...d, status, error } : d)),
    );
  }

  async function addFiles(files: File[]) {
    // 1. Show optimistic "uploading" placeholders with temporary client IDs.
    const placeholders: DocItem[] = files.map((f) => ({
      id: newId(), // temp id, swapped after server replies
      name: f.name,
      size: f.size,
      mime: f.type,
      addedAt: Date.now(),
      status: "uploading",
    }));
    updateDocs((p) => [...placeholders, ...p]);

    await Promise.all(
      files.map(async (file, i) => {
        const temp = placeholders[i];
        try {
          const { document_id } = await documentsApi.upload(file);

          // Replace the placeholder with the server-issued id and "indexing" status.
          updateDocs((p) =>
            p.map((d) =>
              d.id === temp.id
                ? { ...d, id: document_id, status: "indexing" }
                : d,
            ),
          );
        } catch (e) {
          setStatus(temp.id, "failed", errMsg(e, "Upload failed"));
        }
      }),
    );
  }

  async function removeDoc(id: string) {
    try {
      await documentsApi.remove(id);
    } catch (e) {
      setNotice(`Could not remove the file: ${errMsg(e, "unknown error")}`);
      return;
    }
    updateDocs((p) => p.filter((d) => d.id !== id));
    updateSelected((p) => p.filter((x) => x !== id));
  }

  /* ---------- delete confirmation ---------- */

  async function confirmDelete() {
    if (!pending) return;
    const target = pending;
    setPending(null);

    if (target.kind === "doc") {
      await removeDoc(target.id);
      return;
    }

    try {
      // A draft was never saved on the backend, so there is nothing to delete there.
      if (sessions.some((s) => s.id === target.id)) await remove(target.id);
    } catch (e) {
      setNotice(`Could not delete the chat: ${errMsg(e, "unknown error")}`);
      return;
    }
    setInitial((p) => {
      const next = { ...p };
      delete next[target.id];
      return next;
    });
    if (activeId === target.id) startFresh();
  }

  const pendingName =
    pending?.kind === "chat"
      ? (listItems.find((s) => s.id === pending.id)?.title ?? NEW_CHAT)
      : docs.find((d) => d.id === pending?.id)?.name;

  const banner =
    notice ?? (sessionsError ? `Could not load chats: ${sessionsError}` : null);
  const activeMessages = initial[activeId];

  useEffect(() => {
  const pending = docs.filter(
    (d) => d.status === "uploading" || d.status === "indexing",
  );
  if (pending.length === 0) return;

  const timer = setInterval(async () => {
    try {
      const remote = await documentsApi.list();
      updateDocs((prev) =>
        prev.map((d) => {
          const match = remote.find((r) => r.id === d.id);
          if (!match) return d;
          // Server's status wins. Also pull the error message if any.
          return {
            ...d,
            status:
              match.status === "pending" || match.status === "processing"
                ? "indexing"
                : match.status,
            error: match.error,
          };
        }),
      );

      // Auto-select files that just became ready.
      const newlyReady = remote
        .filter((r) => r.status === "ready")
        .map((r) => r.id);
      if (newlyReady.length) {
        updateSelected((p) => [...new Set([...p, ...newlyReady])]);
      }
    } catch {
      /* network hiccup — try again next tick */
    }
  }, 2500);

  return () => clearInterval(timer);
}, [docs, updateDocs, updateSelected]);

  const sidebar = (
    <Tabs defaultValue="sources" className="h-full min-h-0 gap-0">
      <TabsList className="m-3 mb-0 w-[calc(100%-1.5rem)]">
        <TabsTrigger value="sources">Sources</TabsTrigger>
        <TabsTrigger value="chats">Chats</TabsTrigger>
      </TabsList>
      <TabsContent value="sources" className="min-h-0">
        <SourcesPanel
          docs={docs}
          selectedIds={selectedIds}
          onToggle={(id) =>
            updateSelected((p) =>
              p.includes(id) ? p.filter((x) => x !== id) : [...p, id],
            )
          }
          onToggleAll={(checked) =>
            updateSelected(() => (checked ? readyDocs.map((d) => d.id) : []))
          }
          onAddFiles={addFiles}
          onRemove={(id) => setPending({ kind: "doc", id })}
        />
      </TabsContent>
      <TabsContent value="chats" className="min-h-0">
        <ChatsPanel
          chats={listItems}
          activeId={activeId}
          onSelect={openSession}
          onNew={newChat}
          onDelete={(id) => setPending({ kind: "chat", id })}
        />
      </TabsContent>
    </Tabs>
  );

  

  return (
    <div className="flex h-dvh overflow-hidden bg-background">
      <aside className="hidden w-80 shrink-0 border-e bg-sidebar text-sidebar-foreground md:block">
        {sidebar}
      </aside>

      <Sheet open={sidebarOpen} onOpenChange={setSidebarOpen}>
        <SheetContent
          side="left"
          className="w-[85%] bg-sidebar p-0 md:hidden"
          showCloseButton={false}
        >
          <SheetHeader className="sr-only">
            <SheetTitle>Sources and chats</SheetTitle>
          </SheetHeader>
          {sidebar}
        </SheetContent>
      </Sheet>

      <main className="flex min-w-0 flex-1 flex-col">
        {banner && (
          <div className="border-b bg-destructive/10 px-4 py-2 text-sm text-destructive">
            {banner}
          </div>
        )}
        <div className="min-h-0 flex-1">
          {hydrated && activeMessages !== undefined ? (
            <ChatView
              key={activeId}
              sessionId={activeId}
              title={activeSession?.title ?? NEW_CHAT}
              initialMessages={activeMessages}
              user={user}
              selectedDocs={selectedDocs}
              onTurnFinished={onTurnFinished}
              onOpenSidebar={() => setSidebarOpen(true)}
              onRequestDelete={() => setPending({ kind: "chat", id: activeId })}
            />
          ) : (
            <div className="flex h-full items-center justify-center text-muted-foreground">
              <Spinner className="size-5" />
            </div>
          )}
        </div>
      </main>

      <AlertDialog
        open={pending !== null}
        onOpenChange={(open) => !open && setPending(null)}
      >
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>
              {pending?.kind === "doc"
                ? "Remove this source?"
                : "Delete this chat?"}

import type { UIMessage } from "ai";
import type { SessionSummary } from "./types";

/**
 * Session API -> your FastAPI backend, through the existing
 * /api/backend/[...path] proxy (it attaches the Auth.js bearer token).
 *
 * ASSUMED CONTRACT (edit ROUTES / normalisers below to match your backend):
 *   GET    /api/sessions        -> [{ session_id, title, updated_at }]
 *   GET    /api/sessions/{id}   -> { messages: [{ role, content }] }   (or UIMessage[] with `parts`)
 *   DELETE /api/sessions/{id}
 * The proxy maps /api/backend/<path> -> BACKEND_URL/<path>.
 */
const PROXY = "/api/backend";
const ROUTES = {
  list: "/api/sessions",
  one: (id: string) => `/api/sessions/${encodeURIComponent(id)}`,
};

type Raw = Record<string, unknown>;

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${PROXY}${path}`, { cache: "no-store", ...init });
  if (!res.ok) {
    let detail = "";
    try {
      const body = (await res.json()) as Raw;
      detail = typeof body.detail === "string" ? body.detail : "";
    } catch {
      /* non-JSON error body */
    }
    throw new Error(detail || `Request failed (${res.status})`);
  }
  if (res.status === 204) return undefined as T;
  return (await res.json()) as T;
}

/** crypto.randomUUID only exists on localhost/HTTPS; fall back so "New chat" never throws. */
export function newId(): string {
  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
    return crypto.randomUUID();
  }
  const b = crypto.getRandomValues(new Uint8Array(16));
  b[6] = (b[6] & 0x0f) | 0x40;
  b[8] = (b[8] & 0x3f) | 0x80;
  const h = [...b].map((x) => x.toString(16).padStart(2, "0")).join("");
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20)}`;
}

function toMs(v: unknown): number {
  if (typeof v === "number") return v < 1e12 ? v * 1000 : v;
  if (typeof v === "string") {
    const t = Date.parse(v);
    return Number.isNaN(t) ? 0 : t;
  }
  return 0;
}

function toSummary(r: Raw): SessionSummary {
  return {
    id: String(r.session_id ?? r.id ?? r.sessionId ?? ""),
    title: String(r.title ?? r.name ?? "").trim() || "New chat",
    updatedAt: toMs(r.updated_at ?? r.updatedAt ?? r.created_at ?? r.createdAt),
  };
}

function toRole(v: unknown): "user" | "assistant" | null {
  if (v === "user" || v === "human") return "user";
  if (v === "assistant" || v === "ai") return "assistant";
  return null; // system/tool rows are not shown
}

function toUIMessage(r: Raw): UIMessage | null {
  const role = toRole(r.role ?? r.type);
  if (!role) return null;
  const id = String(r.id ?? newId());
  if (Array.isArray(r.parts)) return { id, role, parts: r.parts } as UIMessage;
  const text = typeof r.content === "string" ? r.content : typeof r.text === "string" ? r.text : "";
  return { id, role, parts: [{ type: "text", text }] };
}

export const sessionsApi = {
  async list(): Promise<SessionSummary[]> {
    const data = await request<unknown>(ROUTES.list);
    const rows = Array.isArray(data) ? data : ((data as Raw)?.sessions ?? (data as Raw)?.items ?? []);
    return (rows as Raw[])
      .map(toSummary)
      .filter((s) => s.id)
      .sort((a, b) => b.updatedAt - a.updatedAt);
  },

  async get(id: string): Promise<UIMessage[]> {
    const data = await request<unknown>(ROUTES.one(id));
    const rows = Array.isArray(data) ? data : ((data as Raw)?.messages ?? (data as Raw)?.history ?? []);
    return (rows as Raw[]).map(toUIMessage).filter((m): m is UIMessage => m !== null);
  },

  async remove(id: string): Promise<void> {
    await request<void>(ROUTES.one(id), { method: "DELETE" });
  },
};

"use client";

import { useCallback, useSyncExternalStore } from "react";

/**
 * Tiny localStorage-backed store (SSR safe). Swap for real API calls once
 * the backend exists; components only see `[value, update]`.
 */
const listeners = new Set<() => void>();
const cache = new Map<string, unknown>();

function load<T>(key: string, fallback: T): T {
  if (cache.has(key)) return cache.get(key) as T;
  let value = fallback;
  try {
    const raw = localStorage.getItem(key);
    if (raw) value = JSON.parse(raw) as T;
  } catch {
    /* corrupted or unavailable storage: use fallback */
  }
  cache.set(key, value);
  return value;
}

export function useStored<T>(key: string, fallback: T) {
  const value = useSyncExternalStore(
    (cb) => {
      listeners.add(cb);
      return () => {
        listeners.delete(cb);
      };
    },
    () => load(key, fallback),
    () => fallback,
  );

  const update = useCallback(
    (fn: (prev: T) => T) => {
      const next = fn(load(key, fallback));
      cache.set(key, next);
      try {
        localStorage.setItem(key, JSON.stringify(next));
      } catch {
        /* quota exceeded: keep in-memory value */
      }
      listeners.forEach((l) => l());
    },
    [key, fallback],
  );

  return [value, update] as const;
}

export function useHydrated() {
  return useSyncExternalStore(
    () => () => {},
    () => true,
    () => false,
  );
}

"use client";

import { useCallback, useEffect, useState } from "react";
import { sessionsApi } from "./sessions-api";
import type { SessionSummary } from "./types";

export function useSessions() {
  const [sessions, setSessions] = useState<SessionSummary[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const refresh = useCallback(async () => {
    try {
      setSessions(await sessionsApi.list());
      setError(null);
    } catch (e) {
      setError(e instanceof Error ? e.message : "Could not load chats");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void refresh();
  }, [refresh]);

  const remove = useCallback(async (id: string) => {
    await sessionsApi.remove(id);
    setSessions((p) => p.filter((s) => s.id !== id));
  }, []);

  return { sessions, loading, error, refresh, remove };
}

/**
 * Document API — talks to Next.js proxy routes, which forward to FastAPI.
 *
 *   POST   documents/upload              multipart "file"  → upload + start ingestion
 *   GET    documents/upload              → list this user's files with status
 *   DELETE documents/upload/:id          → delete file + vectors

export interface RemoteDoc {
  id: string;
  filename: string;
  status: "pending" | "processing" | "ready" | "failed";
  size: number;
  error?: string;
  created_at?: string;
}

async function readError(res: Response): Promise<string> {
  try {
    const body = (await res.json()) as { detail?: string };
    return body.detail ?? `Request failed (${res.status})`;
  } catch {
    return `Request failed (${res.status})`;
  }
}

export const documentsApi = {
  async upload(file: File): Promise<{ document_id: string; status: string }> {
    const body = new FormData();
    body.append("file", file);
    const res = await fetch("/api/documents", { method: "POST", body });
    if (!res.ok) throw new Error(await readError(res));
    return (await res.json()) as { document_id: string; status: string };
  },

  async list(): Promise<RemoteDoc[]> {
    const res = await fetch("/api/documents", { cache: "no-store" });
    if (!res.ok) throw new Error(await readError(res));
    const data = (await res.json()) as RemoteDoc[] | { documents: RemoteDoc[] };
    return Array.isArray(data) ? data : data.documents;
  },

  async remove(id: string): Promise<void> {
    const res = await fetch(`/api/documents/${encodeURIComponent(id)}`, {
      method: "DELETE",
    });
    if (!res.ok) throw new Error(await readError(res));
  },
